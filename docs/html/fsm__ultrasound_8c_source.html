170static void do_set_distance(fsm_t *p_this)
171{ // TODO AQUI VIENE LA GORDA
172 fsm_ultrasound_t *p_fsm = (fsm_ultrasound_t *)p_this;
173
174 // 1. Retrieve echo init tick, echo end tick, and echo overflows
175 uint32_t echo_init_tick = port_ultrasound_get_echo_init_tick(p_fsm->ultrasound_id);
176 uint32_t echo_end_tick = port_ultrasound_get_echo_end_tick(p_fsm->ultrasound_id);
177 uint32_t echo_overflows = port_ultrasound_get_echo_overflows(p_fsm->ultrasound_id);
178
179 // 2. Calculate the total time (ticks)
180 uint32_t time_us; // tiempo en microsegundos
181 if (echo_end_tick >= echo_init_tick)
182 {
183 time_us = (echo_end_tick - echo_init_tick) + (echo_overflows * 0xFFFFFFFF);
184 }
185 else
186 {
187 time_us = ((0xFFFFFFFF - echo_init_tick) + echo_end_tick + 1) + (echo_overflows * 0xFFFFFFFF);
188 }
189
190 // 3. Calculate the distance in cm
191
192 uint32_t distance_cm = (time_us * 343) / (2 * 10000);
193
194 // 4. Store the distance in the array
195 p_fsm->distance_arr[p_fsm->distance_idx] = distance_cm;
196
197 // 5. If the array is full, sort it
198 if (p_fsm->distance_idx == FSM_ULTRASOUND_NUM_MEASUREMENTS - 1)
199 {
200 qsort(p_fsm->distance_arr, FSM_ULTRASOUND_NUM_MEASUREMENTS, sizeof(uint32_t), _compare);
201
202 // 6. Compute the median
203 if (FSM_ULTRASOUND_NUM_MEASUREMENTS % 2 == 0)
204 {
205 p_fsm->distance_cm = (p_fsm->distance_arr[FSM_ULTRASOUND_NUM_MEASUREMENTS / 2 - 1] +
206 p_fsm->distance_arr[FSM_ULTRASOUND_NUM_MEASUREMENTS / 2]) /
207 2;
208 }
209 else
210 {
211 p_fsm->distance_cm = p_fsm->distance_arr[FSM_ULTRASOUND_NUM_MEASUREMENTS / 2];
212 }
213
214 // 7. Set the new_measurement flag
215 p_fsm->new_measurement = true;
216 }
217
218 // 8. Increase the index and reset if necessary
219 p_fsm->distance_idx = (p_fsm->distance_idx + 1) % FSM_ULTRASOUND_NUM_MEASUREMENTS;
220
221 // 9. Stop the echo timer
222 port_ultrasound_stop_echo_timer(p_fsm->ultrasound_id);
223
224 // 10. Reset the echo ticks
225 port_ultrasound_reset_echo_ticks(p_fsm->ultrasound_id);
226}
