49void _compute_display_levels(rgb_color_t * p_color, int32_t distance_cm){
50 if (!p_color) return;
51
52 if (distance_cm >= DANGER_MIN_CM && distance_cm <= WARNING_MIN_CM) {
53
54 p_color->r = 255; p_color->g = 0; p_color->b = 0;
55 return;
56 }
57 else if (distance_cm > WARNING_MIN_CM && distance_cm <= NO_PROBLEM_MIN_CM) {
58
59 p_color->r = 237; p_color->g = 150; p_color->b = 0;
60 return;
61 }
62 else if (distance_cm > NO_PROBLEM_MIN_CM && distance_cm <= INFO_MIN_CM) {
63
64 p_color->r = 0; p_color->g = 255; p_color->b = 0;
65 return;
66 }
67 else if (distance_cm > INFO_MIN_CM && distance_cm <= OK_MIN_CM) {
68
69 p_color->r = 26; p_color->g = 89; p_color->b = 82;
70 return;
71 }
72 else if (distance_cm > OK_MIN_CM && distance_cm <= OK_MAX_CM) {
73
74 p_color->r = 0; p_color->g = 0; p_color->b = 255;
75 return;
76 }
77 else if (distance_cm >OK_MAX_CM){
78 // Fuera de rango: apagado
79 p_color->r = 0; p_color->g = 0; p_color->b = 0;
80 return;
81 }
82}
