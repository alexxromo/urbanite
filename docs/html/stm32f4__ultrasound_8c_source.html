207 /* Enalbe clock */
208 RCC->APB1ENR |= RCC_APB1ENR_TIM3EN;
209
210 /* Disable counter CR1*/
211 TIM3->CR1 &= ~TIM_CR1_CEN;
212
213 /* Enable autoreload preload */
214 TIM3->CR1 |= TIM_CR1_ARPE;
215
216 /*Set the counter to 0*/
217 TIM3->CNT = 0;
218
219 /* Compute PSC and ARR */
220 double fclk = (double)SystemCoreClock; /*APB1 = SYSCLK/2*/
221 double t_interr = (double)PORT_PARKING_SENSOR_TRIGGER_UP_US * 1.0e-6; /*Convert µs to seconds*/
222 double arr_max = 65535.0;
223
224 /*Compute initial PSC with ARR max*/
225 double psc_temp = (fclk * t_interr) / (arr_max + 1.0) - 1.0;
226 psc_temp = round(psc_temp); /*Round to Integer*/
227
228 /*Re-compute ARR with the PSC value*/
229 double arr_temp = (fclk * t_interr) / (psc_temp + 1.0) - 1.0;
230 arr_temp = round(arr_temp);
231
232 /* Adjust if ARR exceeds ARR max*/
233
234 while (arr_temp > 65535.0)
235 {
236 psc_temp += 1.0;
237 arr_temp = (fclk * t_interr) / (psc_temp + 1.0) - 1.0;
238 arr_temp = round(arr_temp);
239 }
240
241 /*Load PSC and ARR into registers*/
242 TIM3->PSC = (uint32_t)psc_temp;
243 TIM3->ARR = (uint32_t)arr_temp;
244
245 /*Generate an update event to load preload registers*/
246 TIM3->EGR = TIM_EGR_UG;
247
248 /*Clear the interrupt flag*/
249 TIM3->SR = ~TIM_SR_UIF;

251 /*Enable timer interrupts*/
