96void _timer_pwm_config (uint32_t display_id){//TODO si da fallos cambiar el apb1 por apb2
97 /* 1) Enable clock for TIM4 (APB1) */
98 RCC->APB1ENR |= RCC_APB1ENR_TIM4EN;
99
100 /* 2) Disable counter and enable auto-reload preload */
101 TIM4->CR1 &= ~TIM_CR1_CEN;
102 TIM4->CR1 |= TIM_CR1_ARPE;
103
104 /* 3) Reset counter */
105 TIM4->CNT = 0;
106
107 /* Compute PSC and ARR for 50 Hz (20 ms period) */
108 double fclk = (double)SystemCoreClock;
109 double t_pwm = 1.0 / 50.0; // 50 Hz → 20 ms
110 double arr_max = 65535.0;
111
112 double psc_temp = (fclk * t_pwm) / (arr_max + 1.0) - 1.0;
113 psc_temp = round(psc_temp);
114
115 double arr_temp = (fclk * t_pwm) / (psc_temp + 1.0) - 1.0;
116 arr_temp = round(arr_temp);
117
118 while (arr_temp > arr_max) {
119 psc_temp += 1.0;
120 arr_temp = (fclk * t_pwm) / (psc_temp + 1.0) - 1.0;
121 arr_temp = round(arr_temp);
122 }
123
124 TIM4->PSC = (uint32_t)psc_temp;
125 TIM4->ARR = (uint32_t)arr_temp;
126
127 /* 5) Disable output compare channels */
128 TIM4->CCER &= ~(TIM_CCER_CC1E | TIM_CCER_CC2E | TIM_CCER_CC3E | TIM_CCER_CC4E);
129
130 /* 6) Clear polarity bits */
131 TIM4->CCER &= ~(TIM_CCER_CC1P | TIM_CCER_CC1NP |
132 TIM_CCER_CC2P | TIM_CCER_CC2NP |
133 TIM_CCER_CC3P | TIM_CCER_CC3NP |
134 TIM_CCER_CC4P | TIM_CCER_CC4NP);
135
136 /* 7) Set PWM mode 1 and enable preload for each channel */
137 TIM4->CCMR1 &= ~(TIM_CCMR1_OC1M | TIM_CCMR1_OC2M);
138 TIM4->CCMR1 |= (TIM_CCMR1_OC1M_1 | TIM_CCMR1_OC1M_2 | TIM_CCMR1_OC1PE);
139 TIM4->CCMR1 |= (TIM_CCMR1_OC2M_1 | TIM_CCMR1_OC2M_2 | TIM_CCMR1_OC2PE);
140
141 TIM4->CCMR2 &= ~(TIM_CCMR2_OC3M | TIM_CCMR2_OC4M);
142 TIM4->CCMR2 |= (TIM_CCMR2_OC3M_1 | TIM_CCMR2_OC3M_2 | TIM_CCMR2_OC3PE);
143 TIM4->CCMR2 |= (TIM_CCMR2_OC4M_1 | TIM_CCMR2_OC4M_2 | TIM_CCMR2_OC4PE);
144
145 /* 8) Generate update event to load PSC and ARR */
146 TIM4->EGR |= TIM_EGR_UG;
147
148 /* 9) Disable the timer after configuration */
149 TIM4->CR1 &= ~TIM_CR1_CEN;
150
151}

161 /* 1) Get the display struct */
162 stm32f4_display_hw_t *p_display = _stm32f4_display_get(display_id);
163
164 if (p_display == NULL) {
165 return; // Exit if the display ID is invalid
166 }
167
168 /* 2) Configure the GPIO pins for the RGB LED */
169 stm32f4_system_gpio_config(p_display->p_port_red, p_display->pin_red, STM32F4_GPIO_MODE_AF, STM32F4_GPIO_PUPDR_NOPULL);
170 stm32f4_system_gpio_config(p_display->p_port_green, p_display->pin_green, STM32F4_GPIO_MODE_AF, STM32F4_GPIO_PUPDR_NOPULL);
171 stm32f4_system_gpio_config(p_display->p_port_blue, p_display->pin_blue, STM32F4_GPIO_MODE_AF, STM32F4_GPIO_PUPDR_NOPULL);
172
173 /*Call function stm32f4_system_gpio_config_alternate() with the right arguments to configure the alternate function of the each RGB LED.*/

175 stm32f4_system_gpio_config_alternate(p_display->p_port_green, p_display->pin_green, STM32F4_GPIO_MODE_AF);
176 stm32f4_system_gpio_config_alternate(p_display->p_port_blue, p_display->pin_blue, STM32F4_GPIO_MODE_AF);
177
178 /*Call function _timer_pwm_config() to configure the timer and the PWM signal of the display.*/
179 _timer_pwm_config(display_id);

182 port_display_set_rgb (display_id, COLOR_OFF);

193void port_display_set_rgb (uint32_t display_id, rgb_color_t color){
194
195 /* 1) Leer componentes */
196 uint8_t r = color.r;
197 uint8_t g = color.g;
198 uint8_t b = color.b;
199
200 /* 2) Comprobar display_id */
201 if (display_id != PORT_REAR_PARKING_DISPLAY_ID) {
202 return;
203 }
204
205 /*3) Desactivar el temporizador */
206 TIM4->CR1 &= ~TIM_CR1_CEN;
207
208
209
210 /* 4) Si todos son cero, desactivar todos los canales y saltar a FIN */
211 if (r == 0 && g == 0 && b == 0) {
212 TIM4->CCER &= ~(TIM_CCER_CC1E /* Rojo */
213 | TIM_CCER_CC2E /* Verde */
214 | TIM_CCER_CC4E); /* Azul */
215 /* 7) FIN: forzar actualización*/
216 TIM4->EGR |= TIM_EGR_UG;
217 return;
218}
219
220uint32_t arr = TIM4->ARR; // <-- capturamos ARR
221uint32_t max = arr + 1; // total de pasos
222
223 /* 5) Canal Rojo (CCR1 / CC1E) */
224 if (r == 0) {
225 TIM4->CCER &= ~TIM_CCER_CC1E; // Deshabilitar el canal rojo si el valor es 0
226 } else {
227 TIM4->CCR1 = (uint32_t)r * max / 255; // Asignar el valor de r al registro CCR1
228 TIM4->CCER |= TIM_CCER_CC1E; // Habilitar el canal rojo
229 }
230
231/* 6) Canal Verde (CCR3 / CC3E) */
232if (g == 0) {
233 TIM4->CCER &= ~TIM_CCER_CC3E;
234} else {
235 TIM4->CCR3 = (uint32_t)g * max / 255;
236 TIM4->CCER |= TIM_CCER_CC3E;
237}
238
239/* 7) Canal Azul (CCR4 / CC4E) */
240if (b == 0) {
241 TIM4->CCER &= ~TIM_CCER_CC4E;
242} else {
243 TIM4->CCR4 = (uint32_t)b * max / 255;
244 TIM4->CCER |= TIM_CCER_CC4E;
245}
246
247/* 8) Forzar carga de PSC/ARR e iniciar el timer */
248TIM4->EGR |= TIM_EGR_UG;
249TIM4->CR1 |= TIM_CR1_CEN;
250}
